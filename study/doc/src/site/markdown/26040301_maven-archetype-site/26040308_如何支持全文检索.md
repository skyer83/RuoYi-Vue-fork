# 如何支持全文检索

## 概述

为 Maven Site 文档站点添加全文检索功能，使用 [MiniSearch](https://github.com/lucaong/minisearch) 客户端搜索库实现。

## 实现步骤

### 1. 添加 MiniSearch CDN 引用

在 `pom.xml` 中配置 `maven-replacer-plugin`，将 MiniSearch CDN 脚本注入到所有 HTML 文件：

```xml
<plugin>
  <groupId>com.google.code.maven-replacer-plugin</groupId>
  <artifactId>replacer</artifactId>
  <version>1.5.3</version>
  <executions>
    <!-- 子目录 HTML 文件（先执行） -->
    <execution>
      <id>add-minisearch-subdir</id>
      <phase>site</phase>
      <goals><goal>replace</goal></goals>
      <configuration>
        <basedir>${project.build.directory}/site</basedir>
        <includes>
          <include>*/*.html</include>
          <include>*/*/*.html</include>
        </includes>
        <replacements>
          <replacement>
            <token><![CDATA[</body>]]></token>
            <value><![CDATA[<script src="https://cdn.jsdelivr.net/npm/minisearch@6.3.0/dist/umd/index.min.js"></script>
<script src="../js/site-enhancements.js"></script></body>]]></value>
          </replacement>
        </replacements>
      </configuration>
    </execution>
    <!-- 根目录 HTML 文件（后执行） -->
    <execution>
      <id>add-minisearch-root</id>
      <phase>site</phase>
      <goals><goal>replace</goal></goals>
      <configuration>
        <basedir>${project.build.directory}/site</basedir>
        <includes>
          <include>*.html</include>
        </includes>
        <replacements>
          <replacement>
            <token><![CDATA[</body>]]></token>
            <value><![CDATA[<script src="https://cdn.jsdelivr.net/npm/minisearch@6.3.0/dist/umd/index.min.js"></script>
<script src="./js/site-enhancements.js"></script></body>]]></value>
          </replacement>
        </replacements>
      </configuration>
    </execution>
  </executions>
</plugin>
```

### 3. 创建搜索功能脚本

创建 `src/site/resources/js/site-enhancements.js`，实现以下功能：

#### 3.1 初始化 MiniSearch

```javascript
miniSearch = new MiniSearch({
  fields: ['title', 'content'],
  storeFields: ['title', 'url', 'content'],
  tokenize: function(text) {
    // 自定义分词：支持中文逐字分词，同时保留英文单词和技术术语
    var tokens = [];
    var parts = text.toLowerCase().split(/[^\w\u4e00-\u9fa5]+/)
      .filter(function(t) { return t.length > 0; });
    parts.forEach(function(part) {
      if (/^[\u4e00-\u9fa5]+$/.test(part)) {
        // 纯中文逐字拆分
        for (var i = 0; i < part.length; i++) {
          tokens.push(part[i]);
        }
      } else {
        // 英文/数字作为整体保留
        tokens.push(part);
      }
    });
    return tokens;
  },
  searchOptions: {
    boost: { title: 3, content: 1 },
    fuzzy: function(term) { return term.length > 4 ? 0.2 : 0; },
    prefix: true
  }
});
```

#### 3.2 构建搜索索引

从导航菜单获取所有文档链接，异步获取内容建立索引：

```javascript
function loadSearchIndex() {
  var links = document.querySelectorAll('#navcolumn a[href$=".html"]');
  var documents = [];
  var id = 0;
  
  links.forEach(function(link) {
    var href = link.getAttribute('href').split('#')[0];
    var title = link.textContent.trim();
    documents.push({
      id: id++,
      url: href,
      title: title,
      content: title
    });
  });
  
  // 先添加文档（标题）
  miniSearch.addAll(documents);
  
  // 异步获取内容并更新索引
  documents.forEach(function(doc) {
    fetch(doc.url)
      .then(function(r) { return r.text(); })
      .then(function(html) {
        // 移除图片标签防止404错误
        html = html.replace(/<img[^>]*>/gi, '');
        var tempDiv = document.createElement('div');
        tempDiv.innerHTML = html;
        // 移除无关元素
        var scripts = tempDiv.querySelectorAll(
          'script, style, nav, #banner, #breadcrumbs, #footer'
        );
        scripts.forEach(function(s) { s.remove(); });
        var content = (tempDiv.textContent || '').trim()
          .substring(0, 5000);
        
        // 更新索引
        if (content && content.length > doc.title.length) {
          miniSearch.replace({
            id: doc.id,
            url: doc.url,
            title: doc.title,
            content: content
          });
        }
      });
  });
}
```

#### 3.3 执行搜索

```javascript
function performSearch(query, resultsContainer) {
  var results = miniSearch.search(query, {
    boost: { title: 3, content: 1 },
    fuzzy: 0.3,
    prefix: true
  });
  
  if (results.length === 0) {
    resultsContainer.innerHTML = 
      '<div style="padding:20px;text-align:center;color:#999;">' +
      '未找到匹配结果</div>';
  } else {
    resultsContainer.innerHTML = results.slice(0, 10)
      .map(function(r) {
        return '<div class="search-result-item" data-url="' + 
          r.url + '">' +
          '<div style="font-weight:bold;">' + r.title + '</div>' +
          '</div>';
      }).join('');
  }
}
```

### 4. 关键问题解决方案

#### 4.1 中文分词问题

**问题**：默认分词将连续中文字符作为一个整体，导致搜索单字无法匹配。

**解决**：自定义分词函数，将纯中文字符串逐字拆分：

```javascript
// "项目Git提交规范说明" → ['项', '目', 'git', '提', '交', '规', '范', '说', '明']
```

#### 4.2 图片 404 错误

**问题**：使用 `tempDiv.innerHTML = html` 时，浏览器会尝试加载其中的图片资源。

**解决**：在设置 innerHTML 前移除所有 `<img>` 标签：

```javascript
html = html.replace(/<img[^>]*>/gi, '');
```

#### 4.3 子目录资源路径

**问题**：子目录中的 HTML 文件引用 CSS/JS 路径不正确。

**解决**：使用 replacer 插件分别处理根目录和子目录：
- 子目录使用 `../css/search.css` 和 `../js/site-enhancements.js`
- 根目录使用 `./css/search.css` 和 `./js/site-enhancements.js`

